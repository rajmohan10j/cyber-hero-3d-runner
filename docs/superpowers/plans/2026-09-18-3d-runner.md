# 3D Runner Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a minimalist 3D endless runner mini-game with jump/lanes controls, procedural obstacles, and score tracking.

**Architecture:** Vite + Vanilla JS + Three.js. World moves in -Z direction while player stays stationary. Simple AABB collision detection. localStorage for high scores.

**Tech Stack:** Vite, Three.js, CSS3

## Global Constraints

- Bundle size < 100kb
- 60 FPS desktop / 30 FPS mobile target
- No shadows (performance)
- 3 lanes: positions -3, 0, 3
- Lane width: 3 units
- Jump force: 15 units/sec
- Gravity: 30 units²
- Player: blue cube (#3498db)
- Obstacle: red prism (#e74c3c)
- Score: 1 point per 10 units traveled

- [ ] **Step 5: Write src/scene.js**
```javascript
import * as THREE from 'three';

export function initScene() {
  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0xffffff);

  const camera = new THREE.PerspectiveCamera(
    75,
    window.innerWidth / window.innerHeight,
    0.1,
    1000
  );
  camera.position.set(0, 3, 6);
  camera.lookAt(0, 0, -5);

  const renderer = new THREE.WebGLRenderer({ 
    antialias: true,
    powerPreference: "high-performance"
  });
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  document.getElementById('canvas-container').appendChild(renderer.domElement);

  const light = new THREE.DirectionalLight(0xffffff, 1);
  light.position.set(5, 10, 5);
  scene.add(light);

  scene.add(new THREE.AmbientLight(0xffffff, 0.5));

  const onResize = () => {
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  };
  window.addEventListener('resize', onResize);

  return { scene, camera, renderer };
}

export function createPlayer() {
  const geometry = new THREE.BoxGeometry(1, 1, 1);
  const material = new THREE.MeshLambertMaterial({ color: 0x3498db });
  const player = new THREE.Mesh(geometry, material);
  player.position.y = 0.5;
  return player;
}

export function createGround() {
  const geometry = new THREE.PlaneGeometry(20, 100);
  const material = new THREE.MeshLambertMaterial({ color: 0x95a5a6 });
  const ground = new THREE.Mesh(geometry, material);
  ground.rotation.x = -Math.PI / 2;
  ground.position.z = -20;
  return ground;
}
```

**Files:**

**Interfaces:**
- Consumes: `player` mesh from Task 1
- Produces: `jump()` function, `setLane()` function, `update(delta)` function

- [ ] **Step 1: Write src/player.js**
```javascript
const LANE_WIDTH = 3;
const LANE_POSITIONS = [-LANE_WIDTH, 0, LANE_WIDTH];

const JUMP_FORCE = 15;
const GRAVITY = 30;

let currentLane = 1; // Center lane
let velocityY = 0;
let isJumping = false;

export function setLane(direction) {
  // direction: -1 (left), 0 (stay), 1 (right)
  currentLane = Math.max(0, Math.min(2, currentLane + direction));
}

export function jump() {
  if (!isJumping) {
    isJumping = true;
    velocityY = JUMP_FORCE;
  }
}

export function update(delta) {
  // Update vertical position
  if (isJumping) {
    velocityY -= GRAVITY * delta;
    player.position.y += velocityY * delta;
    
    if (player.position.y <= 0.5) {
      player.position.y = 0.5;
      velocityY = 0;
      isJumping = false;
    }
  }

  // Update horizontal position (smooth lane transition)
  const targetX = LANE_POSITIONS[currentLane];
  player.position.x += (targetX - player.position.x) * 10 * delta;

  // Visual rotation during jump
  if (isJumping) {
    player.rotation.x += 3 * delta;
  }
}

export function getLane() {
  return currentLane;
}

export function reset() {
  currentLane = 1;
  velocityY = 0;
  isJumping = false;
  player.position.set(0, 0.5, 0);
  player.rotation.set(0, 0, 0);
}
```

```javascript
  }
```
```javascript
});
```

### Task 4: Obstacle Spawning & Movement

**Files:**

**Interfaces:**
- Consumes: `scene` from Task 1
- Produces: `spawnObstacle()`, `updateObstacles()`, `resetObstacles()` functions

- [ ] **Step 1: Write src/obstacles.js**
```javascript
import * as THREE from 'three';

const obstacles = [];
const MIN_SPAWN_DISTANCE = 5;
const MAX_SPAWN_DISTANCE = 15;

export function spawnObstacle() {
  const geometry = new THREE.BoxGeometry(1, 
    Math.random() * 3 + 2, // Height: 2-5
    1);
  const material = new THREE.MeshLambertMaterial({ color: 0xe74c3c });
  const obstacle = new THREE.Mesh(geometry, material);
  
  const lane = Math.floor(Math.random() * 3);
  obstacle.position.set(
    [ -3, 0, 3][lane],
    geometry.parameters.height / 2,
    -50 // Spawn far ahead
  );
  
  scene.add(obstacle);
  obstacles.push(obstacle);
  
  return obstacle;
}

export function updateObstacles(speed, deltaTime) {
  // Move all obstacles toward player
  for (let i = obstacles.length - 1; i >= 0; i--) {
    const obs = obstacles[i];
    obs.position.z += speed * deltaTime;
    
    // Remove if passed player
    if (obs.position.z > 2) {
      scene.remove(obs);
      obstacles.splice(i, 1);
    }
  }
}

export function resetObstacles() {
  for (const obs of obstacles) {
    scene.remove(obs);
  }
  obstacles.length = 0;
}

export function getObstacles() {
  return obstacles;
}

export function getNextSpawnDistance() {
  return MIN_SPAWN_DISTANCE + Math.random() * (MAX_SPAWN_DISTANCE - MIN_SPAWN_DISTANCE);
}
```

```javascript
import { getObstacles } from './obstacles.js';

export function checkCollision() {
  const player = scene.children.find(c => c.isMesh && c.material.color?.getHex() === 0x3498db);
  if (!player) return false;
  
  const playerBox = new THREE.Box3().setFromObject(player);
  
  for (const obstacle of getObstacles()) {
    const obstacleBox = new THREE.Box3().setFromObject(obstacle);
    if (playerBox.intersectsBox(obstacleBox)) {
      return true;
    }
  }
  
  return false;
}
```

```javascript
const STORAGE_KEY = '3d-runner-highscore';

export function getHighScore() {
  return parseInt(localStorage.getItem(STORAGE_KEY) || '0');
}

export function saveHighScore(score) {
  const current = getHighScore();
  if (score > current) {
    localStorage.setItem(STORAGE_KEY, score.toString());
  }
}
```
